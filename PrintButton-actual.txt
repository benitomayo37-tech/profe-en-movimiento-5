type PrintDuaPrinciple =
  | "representation"
  | "action-expression"
  | "engagement";

const DUA_PRINT_PALETTE: Record<
  PrintDuaPrinciple,
  {
    background: string;
    text: string;
    border: string;
  }
> = {
  representation: {
    background: "#f5f3ff",
    text: "#6d28d9",
    border: "#8b5cf6",
  },
  "action-expression": {
    background: "#eff6ff",
    text: "#1d4ed8",
    border: "#3b82f6",
  },
  engagement: {
    background: "#ecfdf5",
    text: "#047857",
    border: "#22c55e",
  },
};

function normalizePrintDuaText(text: string): string {
  return text
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .trim()
    .toUpperCase()
    .replace(/^[^A-Z]+/, "");
}

function detectPrintDuaPrinciple(
  text: string,
): PrintDuaPrinciple | null {
  const normalized = normalizePrintDuaText(text);

  if (normalized.startsWith("REPRESENTACION")) {
    return "representation";
  }

  if (
    normalized.startsWith("ACCION Y EXPRESION") ||
    normalized.startsWith("ACCION / EXPRESION")
  ) {
    return "action-expression";
  }

  if (
    normalized.startsWith("COMPROMISO") ||
    normalized.startsWith("MOTIVACION") ||
    normalized.startsWith("IMPLICACION")
  ) {
    return "engagement";
  }

  return null;
}

function isDedicatedDuaSection(
  sectionTitle: string,
): boolean {
  const normalized =
    normalizePrintDuaText(sectionTitle);

  return (
    normalized.includes("ESTRATEGIAS DUA") ||
    normalized.includes("APLICACION DUA")
  );
}

function applyDuaParagraphStyle(
  paragraph: HTMLParagraphElement,
  principle: PrintDuaPrinciple,
  cardStyle: boolean,
): void {
  const palette = DUA_PRINT_PALETTE[principle];

  paragraph.style.setProperty(
    "break-inside",
    "avoid",
    "important",
  );

  paragraph.style.setProperty(
    "page-break-inside",
    "avoid",
    "important",
  );

  if (cardStyle) {
    paragraph.style.setProperty(
      "background-color",
      palette.background,
      "important",
    );

    paragraph.style.setProperty(
      "color",
      palette.text,
      "important",
    );

    paragraph.style.setProperty(
      "border",
      `1px solid ${palette.border}`,
      "important",
    );

    paragraph.style.setProperty(
      "border-left",
      `3px solid ${palette.border}`,
      "important",
    );

    paragraph.style.setProperty(
      "border-radius",
      "6px",
      "important",
    );

    paragraph.style.setProperty(
      "padding",
      "2px 5px",
      "important",
    );

    paragraph.style.setProperty(
      "font-weight",
      "700",
      "important",
    );

    paragraph.style.setProperty(
      "margin-top",
      "2px",
      "important",
    );

    paragraph.style.setProperty(
      "margin-bottom",
      "3px",
      "important",
    );

    return;
  }

  paragraph.style.setProperty(
    "background-color",
    "transparent",
    "important",
  );

  paragraph.style.setProperty(
    "color",
    "#0f172a",
    "important",
  );

  paragraph.style.setProperty(
    "border",
    "0",
    "important",
  );

  paragraph.style.setProperty(
    "border-left",
    `2px solid ${palette.border}`,
    "important",
  );

  paragraph.style.setProperty(
    "padding",
    "0 0 0 5px",
    "important",
  );
}

function applyDuaStylesToPrintClone(
  printClone: HTMLElement,
): void {
  let activePrinciple: PrintDuaPrinciple | null =
    null;

  const rows = Array.from(
    printClone.querySelectorAll<HTMLTableRowElement>(
      ".ai-print-table tbody tr",
    ),
  );

  rows.forEach((row) => {
    const sectionTitle =
      row.querySelector("th")?.textContent ?? "";

    const dedicatedSection =
      isDedicatedDuaSection(sectionTitle);

    const paragraphs = Array.from(
      row.querySelectorAll<HTMLParagraphElement>(
        "td p",
      ),
    );

    paragraphs.forEach((paragraph) => {
      const detectedPrinciple =
        detectPrintDuaPrinciple(
          paragraph.textContent ?? "",
        );

      if (detectedPrinciple) {
        activePrinciple = detectedPrinciple;

        applyDuaParagraphStyle(
          paragraph,
          detectedPrinciple,
          true,
        );

        return;
      }

      if (dedicatedSection && activePrinciple) {
        applyDuaParagraphStyle(
          paragraph,
          activePrinciple,
          false,
        );
      }
    });
  });
}
function waitForImages(
  targetDocument: Document,
): Promise<void> {
  const images = Array.from(targetDocument.images);

  return Promise.all(
    images.map(
      (image) =>
        new Promise<void>((resolve) => {
          if (image.complete) {
            resolve();
            return;
          }

          const finish = () => {
            resolve();
          };

          image.addEventListener("load", finish, {
            once: true,
          });

          image.addEventListener("error", finish, {
            once: true,
          });
        }),
    ),
  ).then(() => undefined);
}

function waitForLayout(
  targetWindow: Window,
): Promise<void> {
  return new Promise((resolve) => {
    targetWindow.requestAnimationFrame(() => {
      targetWindow.requestAnimationFrame(() => {
        resolve();
      });
    });
  });
}

export default function PrintButton() {
  async function handlePrint() {
    const printableArea = document.getElementById(
      "ai-printable-result",
    );

    if (!printableArea) {
      return;
    }

    document.getElementById(PRINT_FRAME_ID)?.remove();

    const printClone =
      printableArea.cloneNode(true) as HTMLElement;

    printClone.id = "ai-print-root";
    printClone.classList.remove("hidden");
    printClone.classList.remove("print:block");
    printClone.removeAttribute("aria-hidden");
    printClone.style.display = "block";
    applyDuaStylesToPrintClone(printClone);

    const iframe = document.createElement("iframe");

    iframe.id = PRINT_FRAME_ID;
    iframe.title = "Documento imprimible de Profe IA";

    Object.assign(iframe.style, {
      position: "fixed",
      top: "0",
      left: "-10000px",
      width: "210mm",
      height: "297mm",
      border: "0",
      pointerEvents: "none",
    });

    document.body.appendChild(iframe);

    const printDocument = iframe.contentDocument;
    const printWindow = iframe.contentWindow;

    if (!printDocument || !printWindow) {
      iframe.remove();
      return;
    }

    printDocument.open();

    printDocument.write(`
      <!doctype html>
      <html lang="es">
        <head>
          <meta charset="utf-8" />
          <title>Profe IA - Documento imprimible</title>
        </head>
        <body></body>
      </html>
    `);

    printDocument.close();

    const base = printDocument.createElement("base");
    base.href = document.baseURI;
    printDocument.head.appendChild(base);

    const style = printDocument.createElement("style");
    style.textContent = PRINT_STYLES;
    printDocument.head.appendChild(style);

    const importedResult = printDocument.importNode(
      printClone,
      true,
    );

    printDocument.body.appendChild(importedResult);

    try {
      await Promise.all([
        waitForImages(printDocument),
        printDocument.fonts.ready,
      ]);

      await waitForLayout(printWindow);
    } catch {
      iframe.remove();
      return;
    }

    let cleaned = false;

    const cleanup = () => {
      if (cleaned) {
        return;
      }

      cleaned = true;

      printWindow.removeEventListener(
        "afterprint",
        cleanup,
      );

      window.removeEventListener(
        "afterprint",
        cleanup,
      );

      iframe.remove();
    };

    printWindow.addEventListener(
      "afterprint",
      cleanup,
      { once: true },
    );

    window.addEventListener(
      "afterprint",
      cleanup,
      { once: true },
    );

    printWindow.focus();
    printWindow.print();
  }

  return (
    <button
      type="button"
      onClick={handlePrint}
      className="inline-flex min-h-10 items-center justify-center gap-2 rounded-xl border border-slate-200 bg-white px-4 py-2 text-sm font-bold text-slate-700 transition hover:border-blue-300 hover:bg-blue-50 hover:text-blue-700 focus-visible:outline-none focus-visible:ring-4 focus-visible:ring-blue-100"
    >
      <span aria-hidden="true">🖨️</span>
      Imprimir
    </button>
  );
}
